
# chatbot_whatsapp.py - Bot WhatsApp Cloud API (Meta Developers)
# Ignora grupos automaticamente - a API oficial ja so entrega 1-a-1,
# mas deixamos a trava extra por seguranca.

from flask import Flask, request, jsonify
import requests
import os

app = Flask(__name__)

# --- COLOQUE AQUI SEUS DADOS DO META DEVELOPERS ---
VERIFY_TOKEN = "MEU_TOKEN_DE_VERIFICACAO"  # o que voce criou no painel
WHATSAPP_TOKEN = "SEU_TOKEN_PERMANENTE_AQUI" # token do Usuario do Sistema
PHONE_NUMBER_ID = "SEU_PHONE_NUMBER_ID" # ID do numero de teste
API_URL = f"https://graph.facebook.com/v21.0/{PHONE_NUMBER_ID}/messages"

def enviar_mensagem(para, texto):
    headers = {
        "Authorization": f"Bearer {WHATSAPP_TOKEN}",
        "Content-Type": "application/json"
    }
    payload = {
        "messaging_product": "whatsapp",
        "to": para,
        "type": "text",
        "text": {"body": texto}
    }
    r = requests.post(API_URL, headers=headers, json=payload)
    print("Enviado:", r.status_code, r.text)
    return r

def responder_logica(remetente, texto_recebido):
    texto = texto_recebido.lower().strip()

    if texto in ["oi", "ola", "olá", "bom dia", "boa tarde", "boa noite", "menu"]:
        return """Fala! 👋 Sou o bot do Cayo.

Digite:
1 - Preço / Orçamento
2 - Horário de atendimento
3 - Falar com humano"""

    elif texto == "1":
        return "Pra orçamento, me manda o que você precisa que já te passo o valor certinho! 💰"
    elif texto == "2":
        return "Atendo de segunda a sexta, das 9h às 18h. Pode mandar mensagem que respondo assim que der! ⏰"
    elif texto == "3":
        return "Beleza, já chamei o Cayo pra falar com você. Ele já já te responde aqui mesmo. 👊"
    else:
        return f"Recebi: '{texto_recebido}' ✅\n\nDigite MENU pra ver as opções."

@app.route("/webhook", methods=["GET"])
def verificar():
    # Verificacao que a Meta faz na hora de cadastrar o webhook
    mode = request.args.get("hub.mode")
    token = request.args.get("hub.verify_token")
    challenge = request.args.get("hub.challenge")
    if mode == "subscribe" and token == VERIFY_TOKEN:
        return challenge, 200
    return "Token invalido", 403

@app.route("/webhook", methods=["POST"])
def receber():
    data = request.get_json()
    print("Recebido:", data)
    try:
        for entry in data.get("entry", []):
            for change in entry.get("changes", []):
                value = change.get("value", {})

                # TRAVA 1: Ignora status (entregue, lido, etc)
                if "messages" not in value:
                    continue

                for msg in value.get("messages", []):
                    # TRAVA 2: Ignora grupos - se vier com group_id ou sem remetente individual, pula
                    if "group_id" in msg or msg.get("type") != "text":
                        print("Ignorado: grupo ou nao-texto")
                        continue

                    remetente = msg.get("from")  # numero de quem mandou
                    texto = msg.get("text", {}).get("body", "")

                    if not remetente or not texto:
                        continue

                    resposta = responder_logica(remetente, texto)
                    enviar_mensagem(remetente, resposta)
    except Exception as e:
        print("Erro:", e)

    return jsonify({"status": "ok"}), 200

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000, debug=True)
